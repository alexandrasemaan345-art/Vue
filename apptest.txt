const {createApp} = Vue;

const API_URL = 'https://as4022_db_user-afterschool.onrender.com';

createApp({
    data(){
        return{
            lessons: [],
            sortBy: 'subject',
            sortOrder: 'asc',
            cart:[],
            showCart: false,
            name:'',
            phone:'',
            orderMessage:''
        };
    },
   mounted() {
    // FETCH A: GET all lessons from backend 
    fetch(`${API_URL}/lessons`)
        .then(response => response.json())
        .then(data => {
            this.lessons = data.map(lesson => ({
                ...lesson,
                id: lesson._id
            }));
        })
        .catch(error => console.error("Error fetching lessons:", error));
    },
  computed: {
    sortedLessons() {
        const key = this.sortBy;
        const dir = this.sortOrder === 'asc' ? 1 : -1;
        return [...this.lessons].sort((a, b) => {
            if (typeof a[key] === 'number') {
                return (a[key] - b[key]) * dir;
            }
            return a[key].localeCompare(b[key]) * dir;
        });
    },
    isNameValid(){
        return /^[A-Za-z ]+$/.test(this.name.trim());
    },
    isPhoneValid(){
        return /^[0-9]+$/.test(this.phone);
    },
    canCheckout(){
        return this.isNameValid && this.isPhoneValid && this.cart.length > 0;
    }
},
methods: {
    addToCart(lesson) {
        if (lesson.spaces > 0) {
            lesson.spaces--;
            this.cart.push(lesson.id);
        }
    },
    removeFromCart(index) {
        const lessonId = this.cart[index];
        const lesson = this.lessons.find(l => l.id === lessonId);
        lesson.spaces++;
        this.cart.splice(index, 1);
    },
    toggleCart(){
        this.showCart = !this.showCart;
        if (!this.showCart){
            this.orderMessage='';
        }
    },
    checkout(){

        const orderData = {

            phone: this.phone,

        };
